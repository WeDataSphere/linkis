```python
class VariableDependencyGraph:
    def __init__(self):
        self.variables = {}           # 所有变量定义
        self.dependencies = {}        # 变量依赖关系
        self.extractors = {}          # 正则提取器映射
        self.usage_chain = []         # 变量使用链
        
    def track_variable(self, var_name, source_request, extraction_pattern=None):
        """追踪变量定义和提取"""
        if extraction_pattern:
            # 这是一个提取的变量
            self.extractors[var_name] = {
                'source': source_request,
                'pattern': extraction_pattern,
                'depends_on': self._find_dependencies(extraction_pattern)
            }
        else:
            #这是一个预定义变量
            self.variables[var_name] = source_request
    
    def build_dependency_chain(self, start_var):
        """构建变量依赖链"""
        chain = [start_var]
        current = start_var
        
        while current in self.extractors:
            dependencies = self.extractors[current]['depends_on']
            if dependencies:
                chain.extend(dependencies)
                current = dependencies[-1]  # 追踪到最后一个依赖
            else:
                break
                
        return chain
```
